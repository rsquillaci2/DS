Tabs

Navegação por abas com conteúdo alternável via Alpine.js.

Formulário de dados pessoais: nome, CPF, data de nascimento e telefone.

Formulário de endereço: CEP, rua, número, complemento, bairro, cidade e estado.

Upload de documentos: RG, CPF, comprovante de residência e foto do veículo.

HTML
<div x-data="{ tab: 'pessoal', tabs: ['pessoal', 'endereco', 'documentos'], moveFocus(dir) { const i = this.tabs.indexOf(this.tab); const next = this.tabs[(i + dir + this.tabs.length) % this.tabs.length]; this.tab = next; this.$nextTick(() =&gt; document.getElementById('tab-' + next)?.focus()); } }">
  <div class="flex border-b border-gray-200 dark:border-gray-700 overflow-x-auto" role="tablist" aria-label="Seções do formulário">
    <button @click="tab = 'pessoal'" @keydown.arrow-right.prevent="moveFocus(1)" @keydown.arrow-left.prevent="moveFocus(-1)" :tabindex="tab === 'pessoal' ? 0 : -1" :class="tab === 'pessoal' ? 'border-primary text-primary' : 'border-transparent text-gray-500 dark:text-gray-400 hover:text-gray-700'" class="px-4 py-2.5 text-sm font-medium border-b-2 transition-colors cursor-pointer -mb-px whitespace-nowrap border-primary text-primary" role="tab" id="tab-pessoal" :aria-selected="tab === 'pessoal'" aria-controls="panel-pessoal" tabindex="0" aria-selected="true">Dados Pessoais</button>
    <button @click="tab = 'endereco'" @keydown.arrow-right.prevent="moveFocus(1)" @keydown.arrow-left.prevent="moveFocus(-1)" :tabindex="tab === 'endereco' ? 0 : -1" :class="tab === 'endereco' ? 'border-primary text-primary' : 'border-transparent text-gray-500 dark:text-gray-400 hover:text-gray-700'" class="px-4 py-2.5 text-sm font-medium border-b-2 transition-colors cursor-pointer -mb-px whitespace-nowrap border-transparent text-gray-500 dark:text-gray-400 hover:text-gray-700" role="tab" id="tab-endereco" :aria-selected="tab === 'endereco'" aria-controls="panel-endereco" tabindex="-1" aria-selected="false">Endereço</button>
    <button @click="tab = 'documentos'" @keydown.arrow-right.prevent="moveFocus(1)" @keydown.arrow-left.prevent="moveFocus(-1)" :tabindex="tab === 'documentos' ? 0 : -1" :class="tab === 'documentos' ? 'border-primary text-primary' : 'border-transparent text-gray-500 dark:text-gray-400 hover:text-gray-700'" class="px-4 py-2.5 text-sm font-medium border-b-2 transition-colors cursor-pointer -mb-px whitespace-nowrap border-transparent text-gray-500 dark:text-gray-400 hover:text-gray-700" role="tab" id="tab-documentos" :aria-selected="tab === 'documentos'" aria-controls="panel-documentos" tabindex="-1" aria-selected="false">Documentos</button>
  </div>
  <div class="p-6">
    <div x-show="tab === 'pessoal'" role="tabpanel" id="panel-pessoal" aria-labelledby="tab-pessoal">
      <p class="text-sm text-gray-600 dark:text-gray-400">Formulário de dados pessoais: nome, CPF, data de nascimento e telefone.</p>
    </div>
    <div x-show="tab === 'endereco'" role="tabpanel" id="panel-endereco" aria-labelledby="tab-endereco" style="display: none;">
      <p class="text-sm text-gray-600 dark:text-gray-400">Formulário de endereço: CEP, rua, número, complemento, bairro, cidade e estado.</p>
    </div>
    <div x-show="tab === 'documentos'" role="tabpanel" id="panel-documentos" aria-labelledby="tab-documentos" style="display: none;">

    </div>
  </div>
</div>
